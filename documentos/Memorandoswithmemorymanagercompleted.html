🍰 Passo a Passo

1️⃣ Inicializar o MemoryManager

// main.cpp (ou qualquer arquivo que inicia o sistema)

#include "../../memorymanager/include/memory_manager_thread.hpp"

// 1. Cria uma instância GLOBAL (na stack!)
static memorymanager::MemoryManagerThread g_memory_manager;

// 2. Inicializa com X GB (1-30)
uint32_t gb = 20;  // 20 GB
if (!g_memory_manager.init(gb)) {
    std::cerr << "❌ Falha ao inicializar MemoryManager!" << std::endl;
    return 1;
}

// 3. Pronto para usar!
std::cout << "✅ MemoryManager initialized with " << gb << "GB" << std::endl;

2️⃣ Integrar o LerConfig

// main.cpp (continuação)

#include "../../lerconfig/include/config.hpp"

// 4. Injeta o MemoryManager no Config (ANTES de carregar!)
LerConfig::Config::setMemoryManager(&g_memory_manager);

// 5. Carrega as configurações
LerConfig::Config& config = LerConfig::Config::getInstance();
if (!config.carregar("/etc/memorandos/config.conf")) {
    std::cerr << "❌ Falha ao carregar configurações!" << std::endl;
    g_memory_manager.shutdown();
    return 1;
}

// 6. Lê as configurações
const char* mensagem = config.getString("boas_vindas", "Padrão");
int memoria = config.getInt("memory", 20);
bool auth = config.getBool("auth_enabled", false);

std::cout << "💬 " << mensagem << std::endl;
std::cout << "📊 Memória configurada: " << memoria << " GB" << std::endl;

3️⃣ Alocar Memória para um Serviço/Thread

// Em qualquer outro .cpp que precise de memória

#include "../../memorymanager/include/memory_manager_thread.hpp"

// Referência para o gerenciador global
extern memorymanager::MemoryManagerThread g_memory_manager;

void meuServico() {
    uint64_t thread_id = 12345;      // ID único da thread
    uint32_t megabytes = 100;        // 100 MB
    uint32_t start_block = 0;
    void* start_addr = nullptr;
    void* end_addr = nullptr;
    char error_msg[256] = {0};
    
    // ALOCA
    if (g_memory_manager.allocate(thread_id, megabytes, 
                                   start_block, start_addr, end_addr, 
                                   error_msg)) {
        std::cout << "✅ Alocado " << megabytes << "MB no bloco " 
                  << start_block << std::endl;
        
        // Usa a memória...
        uint8_t* data = static_cast<uint8_t*>(start_addr);
        size_t size = static_cast<size_t>(megabytes * 1024 * 1024);
        std::memset(data, 0, size);  // Zera a memória
        
        // LIBERA
        if (g_memory_manager.free(thread_id, error_msg)) {
            std::cout << "✅ Memória liberada" << std::endl;
        } else {
            std::cerr << "❌ Falha ao liberar: " << error_msg << std::endl;
        }
    } else {
        std::cerr << "❌ Falha ao alocar: " << error_msg << std::endl;
    }
}

4️⃣ Shutdown Limpo

// main.cpp (antes de encerrar)

// 1. Remove a referência do Config para o MemoryManager
LerConfig::Config::setMemoryManager(nullptr);

// 2. Shutdown do MemoryManager
g_memory_manager.shutdown();
std::cout << "✅ MemoryManager shutdown completo" << std::endl;

// 3. Return
return 0;

⚠️ Regras de Ouro (Nunca Quebrar)

1. Instância GLOBAL
static memorymanager::MemoryManagerThread g_memory_manager; na stack
2. Injeta antes de carregar
setMemoryManager() ANTES de carregar()
3. Remove antes do shutdown
setMemoryManager(nullptr) ANTES de shutdown()
4. Nunca acesse Config pós-shutdown
O Config usa memória do MM
5. Thread ID único por alocação
Cada thread só pode ter 1 empréstimo ativo
6. Nunca use heap (new/malloc)
Use allocate() do MM
7. Não use stack para buffers grandes
Use allocate() do MM
